cmake_minimum_required(VERSION 3.22)
project(battly_imgui_moulberry92 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IMGUI_SOURCE)
    message(FATAL_ERROR "IMGUI_SOURCE is required")
endif()
set(binding "${IMGUI_SOURCE}/imgui-binding")
file(GLOB generated CONFIGURE_DEPENDS "${binding}/build/android-jni/*.cpp")
file(GLOB support CONFIGURE_DEPENDS "${binding}/src/main/native/*.cpp")
set(include_dirs "${binding}/src/main/native" "${binding}/build/android-jni")
set(sources ${generated} ${support})
foreach(module imgui imnodes imgui-node-editor imguizmo implot ImGuiColorTextEdit imgui-knobs)
    file(GLOB module_sources CONFIGURE_DEPENDS "${IMGUI_SOURCE}/include/${module}/*.cpp")
    list(APPEND sources ${module_sources})
    list(APPEND include_dirs "${IMGUI_SOURCE}/include/${module}")
endforeach()
list(APPEND include_dirs "${IMGUI_SOURCE}/include/imgui_club/imgui_memory_editor")
add_library(imgui-moulberry92-java SHARED ${sources})
target_include_directories(imgui-moulberry92-java PRIVATE ${include_dirs})
target_compile_definitions(imgui-moulberry92-java PRIVATE IMGUI_USER_CONFIG="${binding}/src/main/native/imconfig.h")
# The binding uses only JNI 1.6 functions. Request a version supported by both ART and HotSpot.
target_compile_definitions(imgui-moulberry92-java PRIVATE JNI_VERSION_1_8=JNI_VERSION_1_6)
target_link_options(imgui-moulberry92-java PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-Bsymbolic")
